Crear CamaraScreen.jsx
screens/CamaraScreen.jsx — cámara + GPS + guardar foto

Crea el archivo screens/CamaraScreen.jsx. Esta pantalla hace tres cosas al presionar el botón: toma la foto, guarda una copia permanente con expo-file-system y lee el GPS:

¿Por qué expo-file-system? La URI que devuelve la cámara es temporal — se borra al cerrar la app. FileSystem.copyAsync() la copia a documentDirectory/visitas/ que es permanente.
screens/CamaraScreen.jsx
import { useRef, useState } from 'react';
import {
  View, TouchableOpacity, Text, StyleSheet,
  Alert, ActivityIndicator,
} from 'react-native';
import { CameraView, useCameraPermissions } from 'expo-camera';
import * as Location from 'expo-location';
import * as FileSystem from 'expo-file-system';

export default function CamaraScreen({ navigation }) {
  const camaraRef = useRef(null);
  const [cargando, setCargando] = useState(false);
  const [permission, requestPermission] = useCameraPermissions();

  // Pantalla de carga mientras se verifica el permiso
  if (!permission) {
    return (
      <View style={styles.centro}>
        <ActivityIndicator size="large" color="#059669" />
      </View>
    );
  }

  // Pantalla para solicitar permiso de cámara
  if (!permission.granted) {
    return (
      <View style={styles.centro}>
        <Text style={styles.permisoEmoji}>📷</Text>
        <Text style={styles.permisoTexto}>
          Se necesita permiso de cámara para fotografiar el lugar.
        </Text>
        <TouchableOpacity style={styles.permisoBtn} onPress={requestPermission}>
          <Text style={styles.permisoBtnTexto}>Dar permiso de cámara</Text>
        </TouchableOpacity>
      </View>
    );
  }

  const tomarFoto = async () => {
    if (!camaraRef.current || cargando) return;
    setCargando(true);
    try {
      // 1. Tomar foto
      const foto = await camaraRef.current.takePictureAsync({ quality: 0.7 });

      // 2. Copiar a carpeta permanente para que no se borre
      const dirVisitas = FileSystem.documentDirectory + 'visitas/';
      const dirInfo = await FileSystem.getInfoAsync(dirVisitas);
      if (!dirInfo.exists) {
        await FileSystem.makeDirectoryAsync(dirVisitas, { intermediates: true });
      }
      const rutaPermanente = dirVisitas + 'visita_' + Date.now() + '.jpg';
      await FileSystem.copyAsync({ from: foto.uri, to: rutaPermanente });

      // 3. Leer coordenadas GPS
      const { status } = await Location.requestForegroundPermissionsAsync();
      let lat = null, lng = null;
      if (status === 'granted') {
        const ubicacion = await Location.getCurrentPositionAsync({
          accuracy: Location.Accuracy.Balanced,
        });
        lat = ubicacion.coords.latitude;
        lng = ubicacion.coords.longitude;
      } else {
        Alert.alert('GPS no disponible', 'La foto se guardará sin coordenadas.');
      }

      // 4. Navegar de regreso a Visitas con los datos
      navigation.navigate('Visitas', {
        fotoUri:   rutaPermanente,
        lat, lng,
        fecha:     new Date().toLocaleString('es-MX'),
        timestamp: Date.now(),
      });

    } catch (error) {
      Alert.alert('Error', 'No se pudo tomar la foto. Inténtalo de nuevo.');
      console.error(error);
    } finally {
      setCargando(false);
    }
  };

  // Visor de la cámara
  // IMPORTANTE: CameraView no acepta hijos — el overlay va en un View separado
  return (
    <View style={{ flex: 1, backgroundColor: '#000' }}>

      <CameraView
        style={StyleSheet.absoluteFillObject}
        ref={camaraRef}
        facing="back"
      />

      {/* Overlay encima de la cámara con position absolute */}
      <View style={StyleSheet.absoluteFillObject} pointerEvents="box-none">

        <View style={styles.gpsIndicador}>
          <Text style={styles.gpsTexto}>GPS activo</Text>
        </View>

        {/* Retícula central */}
        <View style={styles.reticula}>
          <View style={[styles.esquina, styles.esqTL]} />
          <View style={[styles.esquina, styles.esqTR]} />
          <View style={[styles.esquina, styles.esqBL]} />
          <View style={[styles.esquina, styles.esqBR]} />
        </View>

        {/* Botón de captura */}
        <View style={styles.botonesArea}>
          <Text style={styles.instruccion}>
            Toca el botón para fotografiar y registrar coordenadas GPS
          </Text>
          <TouchableOpacity
            style={[styles.btnCaptura, cargando && { opacity: 0.6 }]}
            onPress={tomarFoto}
            disabled={cargando}
          >
            {cargando
              ? <ActivityIndicator color="#059669" size="large" />
              : <View style={styles.circuloInterior} />
            }
          </TouchableOpacity>
        </View>

      </View>
    </View>
  );
}

Crear LugaresVisitadosScreen.jsx
screens/LugaresVisitadosScreen.jsx — lista con foto, coordenadas y fecha

Crea el archivo screens/LugaresVisitadosScreen.jsx. Muestra la lista de visitas, guarda los datos con AsyncStorage para que persistan aunque cierres la app, y permite borrar registros:

screens/LugaresVisitadosScreen.jsx
import { useState, useCallback } from 'react';
import {
  View, Text, FlatList, Image, TouchableOpacity,
  StyleSheet, Alert,
} from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { useFocusEffect } from '@react-navigation/native';

const CLAVE_STORAGE = 'lugares_visitados_v1';

export default function LugaresVisitadosScreen({ navigation, route }) {
  const [visitas, setVisitas] = useState([]);

  // Se ejecuta cada vez que la pantalla recibe el foco (al regresar de la cámara)
  useFocusEffect(
    useCallback(() => {
      cargarVisitas().then((actuales) => {
        if (route.params?.fotoUri && route.params?.timestamp) {
          const nueva = {
            id:      String(route.params.timestamp),
            fotoUri: route.params.fotoUri,
            lat:     route.params.lat,
            lng:     route.params.lng,
            fecha:   route.params.fecha,
          };
          const existe = actuales.find((v) => v.id === nueva.id);
          if (!existe) {
            const actualizadas = [nueva, ...actuales];
            persistir(actualizadas);
            setVisitas(actualizadas);
          }
          navigation.setParams({ fotoUri: undefined, timestamp: undefined });
        } else {
          setVisitas(actuales);
        }
      });
    }, [route.params?.timestamp])
  );

  const cargarVisitas = async () => {
    try {
      const json = await AsyncStorage.getItem(CLAVE_STORAGE);
      return json ? JSON.parse(json) : [];
    } catch (e) {
      console.error('Error al leer AsyncStorage:', e);
      return [];
    }
  };

  const persistir = async (datos) => {
    try {
      await AsyncStorage.setItem(CLAVE_STORAGE, JSON.stringify(datos));
    } catch (e) {
      console.error('Error al guardar en AsyncStorage:', e);
    }
  };

  const borrarVisita = (id) => {
    Alert.alert('Borrar visita', '¿Eliminar este registro?', [
      { text: 'Cancelar', style: 'cancel' },
      {
        text: 'Borrar', style: 'destructive',
        onPress: () => {
          const filtradas = visitas.filter((v) => v.id !== id);
          setVisitas(filtradas);
          persistir(filtradas);
        },
      },
    ]);
  };

  const borrarTodo = () => {
    Alert.alert('Borrar todo', '¿Eliminar todas las visitas?', [
      { text: 'Cancelar', style: 'cancel' },
      {
        text: 'Borrar todo', style: 'destructive',
        onPress: async () => {
          await AsyncStorage.removeItem(CLAVE_STORAGE);
          setVisitas([]);
        },
      },
    ]);
  };

  const renderItem = ({ item }) => (
    <View style={styles.card}>
      <Image source={{ uri: item.fotoUri }} style={styles.foto} />
      <View style={styles.cardInfo}>
        {item.lat != null ? (
          <>
            <Text style={styles.coordsLabel}>Coordenadas GPS</Text>
            <Text style={styles.coords}>
              {item.lat.toFixed(5)}, {item.lng.toFixed(5)}
            </Text>
          </>
        ) : (
          <Text style={styles.sinGps}>Sin coordenadas GPS</Text>
        )}
        <Text style={styles.fecha}>{item.fecha}</Text>
      </View>
      <TouchableOpacity
        style={styles.btnBorrarItem}
        onPress={() => borrarVisita(item.id)}
        hitSlop={{ top: 8, bottom: 8, left: 8, right: 8 }}
      >
        <Text style={styles.btnBorrarItemTexto}>✕</Text>
      </TouchableOpacity>
    </View>
  );

  return (
    <View style={styles.contenedor}>

      <TouchableOpacity
        style={styles.btnCamara}
        onPress={() => navigation.navigate('Camara')}
        activeOpacity={0.85}
      >
        <Text style={styles.btnCamaraTexto}>📷  Fotografiar este lugar</Text>
        <Text style={styles.btnCamaraSub}>Guarda foto + coordenadas GPS</Text>
      </TouchableOpacity>

      {visitas.length > 0 && (
        <Text style={styles.contador}>
          {visitas.length} lugar{visitas.length !== 1 ? 'es' : ''} registrado{visitas.length !== 1 ? 's' : ''}
        </Text>
      )}

      <FlatList
        data={visitas}
        keyExtractor={(item) => item.id}
        renderItem={renderItem}
        contentContainerStyle={{ paddingBottom: 24 }}
        ListEmptyComponent={
          <View style={styles.vacio}>
            <Text style={styles.vacioEmoji}>📷</Text>
            <Text style={styles.vacioTitulo}>Sin registros aún</Text>
            <Text style={styles.vacioTexto}>
              Toca el botón verde para abrir la cámara,{'\n'}
              tomar una foto y registrar tus coordenadas GPS.
            </Text>
          </View>
        }
        ListFooterComponent={
          visitas.length > 1 ? (
            <TouchableOpacity style={styles.btnBorrarTodo} onPress={borrarTodo}>
              <Text style={styles.btnBorrarTodoTexto}>Borrar todos los registros</Text>
            </TouchableOpacity>
          ) : null
        }
      />

    </View>
  );
}

  btnCamaraTexto:     { color: '#fff', fontWeight: '800', fontSize: 16 },

App.js — versión completa
import { NavigationContainer }        from '@react-navigation/native';
import { createNativeStackNavigator } from '@react-navigation/native-stack';
import { createBottomTabNavigator }   from '@react-navigation/bottom-tabs';
import { Text, TouchableOpacity }     from 'react-native';

import HomeScreen             from './screens/HomeScreen';
import DetalleScreen          from './screens/DetalleScreen';
import MapaScreen             from './screens/MapaScreen';
import FavoritosScreen        from './screens/FavoritosScreen';
import AcercaScreen           from './screens/AcercaScreen';
import CamaraScreen           from './screens/CamaraScreen';           // NUEVO
import LugaresVisitadosScreen from './screens/LugaresVisitadosScreen'; // NUEVO

const Stack = createNativeStackNavigator();
const Tab   = createBottomTabNavigator();

const TabIcon = ({ emoji, focused }) => (
  <Text style={{ fontSize: 20, opacity: focused ? 1 : 0.5 }}>{emoji}</Text>
);

// Stack: Home → Detalle → Camara → Acerca
function InicioStack() {
  return (
    <Stack.Navigator
      screenOptions={{
        headerStyle:      { backgroundColor: '#065f46' },
        headerTintColor:  '#fff',
        headerTitleStyle: { fontWeight: '700' },
      }}
    >
      <Stack.Screen
        name="Home"
        component={HomeScreen}
        options={({ navigation }) => ({
          title: 'Lugares Misteriosos',
          headerRight: () => (
            <TouchableOpacity onPress={() => navigation.navigate('Acerca')} style={{ marginRight: 4 }}>
              <Text style={{ fontSize: 22 }}>ℹ️</Text>
            </TouchableOpacity>
          ),
        })}
      />
      <Stack.Screen
        name="Detalle"
        component={DetalleScreen}
        options={({ route }) => ({ title: route.params.lugar.nombre })}
      />
      <Stack.Screen
        name="Camara"
        component={CamaraScreen}
        options={{
          title: 'Tomar foto',
          headerStyle:     { backgroundColor: '#000' },
          headerTintColor: '#fff',
        }}
      />
      <Stack.Screen
        name="Acerca"
        component={AcercaScreen}
        options={{ title: 'Acerca de' }}
      />
    </Stack.Navigator>
  );
}

// Tab: Inicio | Mapa | Favoritos | Visitas
function MisTabs() {
  return (
    <Tab.Navigator
      screenOptions={{
        tabBarActiveTintColor:   '#059669',
        tabBarInactiveTintColor: '#aaa',
        headerShown:             false,
      }}
    >
      <Tab.Screen name="Inicio" component={InicioStack}
        options={{ tabBarLabel: 'Inicio', tabBarIcon: ({ focused }) => <TabIcon emoji="🏠" focused={focused} /> }}
      />
      <Tab.Screen name="Mapa" component={MapaScreen}
        options={{
          tabBarLabel: 'Mapa',
          tabBarIcon: ({ focused }) => <TabIcon emoji="🗺️" focused={focused} />,
          headerShown: true, headerTitle: 'Mapa',
          headerStyle: { backgroundColor: '#065f46' }, headerTintColor: '#fff', headerTitleStyle: { fontWeight: '700' },
        }}
      />
      <Tab.Screen name="Favoritos" component={FavoritosScreen}
        options={{
          tabBarLabel: 'Favoritos',
          tabBarIcon: ({ focused }) => <TabIcon emoji="⭐" focused={focused} />,
          headerShown: true, headerTitle: 'Favoritos',
          headerStyle: { backgroundColor: '#065f46' }, headerTintColor: '#fff', headerTitleStyle: { fontWeight: '700' },
        }}
      />
      <Tab.Screen name="Visitas" component={LugaresVisitadosScreen}
        options={{
          tabBarLabel: 'Mis Visitas',
          tabBarIcon: ({ focused }) => <TabIcon emoji="📷" focused={focused} />,
          headerShown: true, headerTitle: 'Lugares Visitados',
          headerStyle: { backgroundColor: '#065f46' }, headerTintColor: '#fff', headerTitleStyle: { fontWeight: '700' },
        }}
      />
    </Tab.Navigator>
  );
}

export default function App() {
  return (
    <NavigationContainer>
      <MisTabs />
    </NavigationContainer>
  );
}
